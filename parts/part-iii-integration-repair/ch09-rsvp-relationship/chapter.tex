\chapter{Relationship to RSVP}
\label{ch:chapter-09-rsvp-relationship}

\begin{definition}[Admissible Reconstruction]
A reconstruction \(x\in\stateSpace\) is admissible if it is consistent with observations in \(\observationSpace\) and satisfies all active hard constraints in \(\constraintSet\).
\end{definition}

\section{Core Objectives}
\begin{itemize}
  \item Map CLIO reconstruction to RSVP projection layers.
  \item Interpret admissibility geometrically in field-state descriptions.
  \item Relate projection entropy to inverse reconstruction quality.
\end{itemize}
