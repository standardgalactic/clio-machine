\chapter{Repair as Constraint Reconfiguration}
\label{ch:chapter-10-repair-reconfiguration}

\begin{definition}[Admissible Reconstruction]
A reconstruction \(x\in\stateSpace\) is admissible if it is consistent with observations in \(\observationSpace\) and satisfies all active hard constraints in \(\constraintSet\).
\end{definition}

\section{Core Objectives}
\begin{itemize}
  \item Use reconstruction failure to diagnose incompatible assumptions.
  \item Define repair operators distinct from optimization steps.
  \item Show iterative knowledge refinement via admissibility-preserving updates.
\end{itemize}
