\chapter{Future Research}
\label{ch:chapter-13-future-research}

\section{Chapter Preface}
Foundational theory should open research programs, not close them. Once primitives are clear, new questions become sharper: hierarchical depth, temporal admissibility, multi-agent feasibility, and categorical structure.

This chapter treats future work as an extension of the same formal spine rather than a speculative epilogue. Each direction is stated in mathematically checkable terms.

The intended outcome is a map of rigorous next steps that preserves continuity with the framework already established.

\section{Derivations and Formal Results}

\begin{definition}[Hierarchical admissibility]
\[
A^{(0)}\supseteq A^{(1)}\supseteq\cdots\supseteq A^{(k)}.
\]
\end{definition}

\begin{proposition}[Finite hierarchical existence preservation]
If each refinement set is nonempty and compact, then every finite refinement level remains nonempty and compact, preserving existence of admissible candidates.
\end{proposition}

\begin{definition}[Temporal CLIO]
For trajectories \(\gamma:[0,T]\to\stateSpace\), path admissibility is
\[
\gamma\in\mathcal A\iff c_i(\gamma,t)\le0\quad\forall t\in[0,T],\ \forall i.
\]
\end{definition}

\begin{definition}[Multi-agent admissibility]
Given agent-specific sets \(A_1,\ldots,A_n\), collective feasibility requires \(\bigcap_iA_i\neq\varnothing\).
\end{definition}

\begin{proposition}[Categorical lifting view]
Observation morphisms induce commutative squares and reconstruction corresponds to constrained lift existence and uniqueness questions.
\end{proposition}
