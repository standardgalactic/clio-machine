\chapter{Conclusion}
\label{ch:chapter-14-conclusion}

\begin{definition}[Admissible Reconstruction]
A reconstruction \(x\in\stateSpace\) is admissible if it is consistent with observations in \(\observationSpace\) and satisfies all active hard constraints in \(\constraintSet\).
\end{definition}

\section{Core Objectives}
\begin{itemize}
  \item Summarize CLIO as admissible reconstruction theory.
  \item Reassert optimization as subordinate to admissibility.
  \item Position CLIO between machine learning, symbolic reasoning, and repair frameworks.
\end{itemize}
