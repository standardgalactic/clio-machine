\chapter{Constraints as Primary Computational Objects}
\label{ch:chapter-03-constraints-primary}

\begin{definition}[Admissible Reconstruction]
A reconstruction \(x\in\stateSpace\) is admissible if it is consistent with observations in \(\observationSpace\) and satisfies all active hard constraints in \(\constraintSet\).
\end{definition}

\section{Core Objectives}
\begin{itemize}
  \item Formalize hard and soft constraints and their interaction topology.
  \item Define compatibility graphs and admissible regions.
  \item Explain consistency-driven emergence of solutions.
\end{itemize}
