\chapter{Introduction}
\label{ch:chapter-01-introduction}

\begin{definition}[Admissible Reconstruction]
A reconstruction \(x\in\stateSpace\) is admissible if it is consistent with observations in \(\observationSpace\) and satisfies all active hard constraints in \(\constraintSet\).
\end{definition}

\section{Core Objectives}
\begin{itemize}
  \item Motivate limitations of prediction-centric machine learning.
  \item Define reconstruction from incomplete observations as the primary inference problem.
  \item State CLIO thesis: computation as incompatibility reduction across interacting constraints.
\end{itemize}
