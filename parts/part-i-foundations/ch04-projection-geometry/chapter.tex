\chapter{Projection Geometry}
\label{ch:chapter-04-projection-geometry}

\begin{definition}[Admissible Reconstruction]
A reconstruction \(x\in\stateSpace\) is admissible if it is consistent with observations in \(\observationSpace\) and satisfies all active hard constraints in \(\constraintSet\).
\end{definition}

\section{Core Objectives}
\begin{itemize}
  \item Treat observations as quotient maps over high-dimensional state spaces.
  \item Characterize invariant retention and information loss.
  \item Define projection entropy and identify reconstruction limits.
\end{itemize}
