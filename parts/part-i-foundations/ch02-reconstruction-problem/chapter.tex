\chapter{The Reconstruction Problem}
\label{ch:chapter-02-reconstruction-problem}

\begin{definition}[Admissible Reconstruction]
A reconstruction \(x\in\stateSpace\) is admissible if it is consistent with observations in \(\observationSpace\) and satisfies all active hard constraints in \(\constraintSet\).
\end{definition}

\section{Core Objectives}
\begin{itemize}
  \item Model observations as partial projections of latent states.
  \item Describe ambiguity classes and structural equivalence under observation.
  \item Introduce admissible reconstruction sets and non-uniqueness conditions.
\end{itemize}
