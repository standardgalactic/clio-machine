\chapter{Optimisation Inside Admissible Space}
\label{ch:chapter-06-optimization-admissible-space}

\begin{definition}[Admissible Reconstruction]
A reconstruction \(x\in\stateSpace\) is admissible if it is consistent with observations in \(\observationSpace\) and satisfies all active hard constraints in \(\constraintSet\).
\end{definition}

\section{Core Objectives}
\begin{itemize}
  \item Separate admissibility from objective optimization.
  \item Define objective functions over feasible manifolds only.
  \item Discuss multiplicity of admissible optima and energy landscapes.
\end{itemize}
